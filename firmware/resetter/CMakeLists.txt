cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(resetter C CXX ASM)
pico_sdk_init()
foreach(target resetter compact-resetter)
  add_executable(${target} main.c usb.c)
  target_link_libraries(${target} pico_stdlib hardware_adc pico_unique_id)
  target_compile_definitions(${target} PRIVATE PICO_STDIO_USB_USE_DEFAULT_DESCRIPTORS=0
    PICO_ENABLE_USB_RESET_VIA_VENDOR_INTERFACE=0)
  if(target STREQUAL "compact-resetter")
    set(version "2.0.0")
    target_link_libraries(${target} hardware_pwm)
    target_compile_definitions(${target} PRIVATE RESETTER_DISCRETE_RGB=1)
  else()
    set(version "1.0.0")
    target_link_libraries(${target} pico_status_led)
    target_compile_definitions(${target} PRIVATE PICO_DEFAULT_WS2812_PIN=29 PICO_STATUS_LED_AVAILABLE=0)
  endif()
  target_compile_definitions(${target} PRIVATE RESETTER_VERSION="${version}")
  pico_enable_stdio_usb(${target} 1)
  pico_enable_stdio_uart(${target} 0)
  pico_set_program_name(${target} "tscircuit USB resetter")
  pico_set_program_version(${target} "${version}")
  pico_add_extra_outputs(${target})
endforeach()
